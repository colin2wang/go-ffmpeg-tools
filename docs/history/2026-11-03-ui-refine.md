>
> 新增需求全部落地：
> ✅ 文件选择 + **拖拽上传（输入视频、字幕轨道）**
> ✅ 输出路径选项：【自定义文件 / 和输入视频同目录 / 程序执行目录】
> ✅ 深色 / 浅色主题一键切换

> ✅ 样式延续专业工具风格，主题变量自动切换

<script setup lang="ts">
import { ref, reactive, computed, watch } from "vue";
import type { SubTrack, SubConfig, Job } from "@/types/task";
import { SubmitJob, CancelJob } from "@/wailsjs/go/main/App";
import { EventsOn } from "@/wailsjs/runtime/runtime";
import { v4 as uuidv4 } from "uuid";

// ========== 主题切换 ==========
const isDark = ref(true);
const themeClass = computed(() => isDark.value ? "theme-dark" : "theme-light");
function toggleTheme() {
  isDark.value = !isDark.value;
}

// ========== 拖拽标记 ==========
const videoDragOver = ref(false);
const subDragOver = ref(false);
const burnSubDragOver = ref(false);

// ========== 表单状态 ==========
const inputVideo = ref("");
// 输出模式：custom / same / exe
const outputMode = ref<"custom" | "same" | "exe">("custom");
const outputVideo = ref("");
const outputSuffix = ref(".subbed.mp4");

const subMode = ref<"soft" | "burn">("soft");
const subTracks = ref<SubTrack[]>([]);

// 硬字幕参数
const burnSubFile = ref("");
const burnFontName = ref("微软雅黑");
const burnFontSize = ref(24);

const isProcessing = ref(false);
const jobList = ref<Job[]>([]);
const showLogMap = reactive<Record<string, boolean>>({});

const statusTextMap: Record<string, string> = {
  pending: "等待中",
  running: "运行中",
  done: "已完成",
  error: "失败",
  cancelled: "已取消",
};

// ========== 拖拽事件 ==========
function onVideoDrop(e: DragEvent) {
  videoDragOver.value = false;
  const file = e.dataTransfer?.files[0];
  if (!file) return;
  inputVideo.value = file.path;
}
function onSubDrop(e: DragEvent) {
  subDragOver.value = false;
  const files = Array.from(e.dataTransfer?.files ?? []);
  files.forEach(file => {
    if (file.name.endsWith(".srt")) {
      subTracks.value.push({
        id: uuidv4(),
        file: file.path,
        lang: "chi",
        title: "中文",
      });
    }
  })
}
function onBurnSubDrop(e: DragEvent) {
  burnSubDragOver.value = false;
  const file = e.dataTransfer?.files[0];
  if (!file) return;
  burnSubFile.value = file.path;
}

// ========== 字幕轨道操作 ==========
function addTrack() {
  subTracks.value.push({
    id: uuidv4(),
    file: "",
    lang: "chi",
    title: "中文",
  });
}
function removeTrack(idx: number) {
  subTracks.value.splice(idx, 1);
}
function moveTrackUp(idx: number) {
  if (idx <= 0) return;
  [subTracks.value[idx - 1], subTracks.value[idx]] = [subTracks.value[idx], subTracks.value[idx - 1]];
}
function moveTrackDown(idx: number) {
  if (idx >= subTracks.value.length - 1) return;
  [subTracks.value[idx + 1], subTracks.value[idx]] = [subTracks.value[idx], subTracks.value[idx + 1]];
}

// ========== 文件选择（对接Wails runtime dialog） ==========
async function selectInputVideo() {
  // const res = await runtime.OpenFileDialog({...})
}
async function selectOutputVideo() {
  // const res = await runtime.SaveFileDialog({...})
}
async function selectBurnSub() {
  // const res = await runtime.OpenFileDialog({...})
}

// ========== 任务提交 ==========
async function submitJob() {
  if (!inputVideo.value) return;
  if (subMode.value === "soft" && subTracks.value.length === 0) return;
  if (subMode.value === "burn" && !burnSubFile.value) return;

  isProcessing.value = true;
  const cfg: SubConfig = {
    mode: subMode.value,
    tracks: subMode.value === "soft" ? [...subTracks.value] : [{id:uuidv4(), file: burnSubFile.value, lang:"chi", title:""}],
    fontName: burnFontName.value,
    fontSize: burnFontSize.value,
  };

  try {
    await SubmitJob({
      inputVideo: inputVideo.value,
      outputVideo: outputVideo.value,
      outputMode: outputMode.value,
      outputSuffix: outputSuffix.value,
      subConfig: cfg,
    });
  } catch (e) {
    console.error(e);
  }
}

// 订阅后端 jobUpdate 事件
EventsOn("jobUpdate", (job: Job) => {
  const exist = jobList.value.find(j => j.jobId === job.jobId);
  if (exist) {
    Object.assign(exist, job);
  } else {
    jobList.value.push(job);
  }
  if (job.status !== "running") {
    isProcessing.value = false;
  }
});

function toggleLog(jobId: string) {
  showLogMap[jobId] = !showLogMap[jobId];
}

// 工具函数
function getFileName(path: string) {
  const sp = path.split("\\");
  return sp.at(-1) ?? path;
}
function formatTime(sec: number) {
  const h = Math.floor(sec / 3600);
  const m = Math.floor((sec % 3600) / 60);

## 4. 后端Go侧适配说明（`app.go`）

你需要修改 `SubmitJob` 函数，拿到 `outputMode / outputSuffix`，在Go里自动拼接输出路径：

- `same`：取输入视频目录 + 原文件名 + suffix
- `exe`：取程序可执行文件目录 + 原文件名 + suffix>
> 示例逻辑（Go伪代码）

```
var outPath string
switch req.OutputMode {
case "same":
    dir := filepath.Dir(req.InputVideo)
    base := strings.TrimSuffix(filepath.Base(req.InputVideo), filepath.Ext(req.InputVideo))
    outPath = filepath.Join(dir, base + req.OutputSuffix)
case "exe":
    exePath, _ := os.Executable()
    dir := filepath.Dir(exePath)
    base := strings.TrimSuffix(filepath.Base(req.InputVideo), filepath.Ext(req.InputVideo))
    outPath = filepath.Join(dir, base + req.OutputSuffix)
default:
    outPath = req.OutputVideo
}
```
