# 项目名称：`go-subburn`
> 含义：go + subtitle burn / mux（既支持烧录硬字幕，也支持多路软字幕封装），符合 `go-` 前缀，简洁好记、GitHub 风格。
> 技术栈：**Go + Wails v2 + Vue3 + TypeScript + Vite**
> 核心能力：
> 1. MP4 多路 `mov_text` 软字幕封装（多语言字幕轨道，播放器可切换）
> 2. 字幕烧录（硬字幕，ass/srt，支持字体自定义）
> 3. 任务队列、实时进度、取消任务、日志输出
> 4. 前端 TS + Vue 配置字幕轨道名称、语言标签、样式
> 5. 内置 ffmpeg 二进制分发方案

> 重要前置：`mov_text` 本身不支持 ASS 特效，**多路软字幕只能是简单文本字幕（srt → mov_text）**；ASS 样式字幕只能走硬烧录。多路软字幕就是多次 `-i xxx.srt -c:s mov_text -metadata:s:s:0 language=eng -metadata:s:s:0 title=English` 追加字幕流。

## 任务管理器 `internal/task/manager.go`
```go
package task

import (
	"sync"
)

type Manager struct {
	mu   sync.Mutex
	jobs map[string]*Job
}

func NewManager() *Manager {
	return &Manager{
		jobs: make(map[string]*Job),
	}
}

func (m *Manager) AddJob(job *Job) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.jobs[job.JobId] = job
}

func (m *Manager) GetJob(id string) *Job {
	m.mu.Lock()
	defer m.mu.Unlock()
	return m.jobs[id]
}

func (m *Manager) UpdateJob(job *Job) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.jobs[job.JobId] = job
}

func (m *Manager) CancelJob(jobId string) bool {
	m.mu.Lock()
	job, ok := m.jobs[jobId]
	m.mu.Unlock()
	if !ok || job.Status != "running" {
		return false
	}
	job.cancel()
	return true
}
```

# 四、FFmpeg 命令构造器 `internal/ffmpeg/builder.go`
> ✅ 重点：**多路软字幕轨道构建逻辑**
> 对于 N 条字幕：
> `-i video.mp4`
> `-i track0.srt -c:s mov_text -metadata:s:s:0 language=zh -metadata:s:s:0 title=中文`
> `-i track1.srt -c:s mov_text -metadata:s:s:1 language=eng -metadata:s:s:1 title=English`
> `-c:v copy -c:a copy` 流复制，不重编码

```go
package ffmpeg

import (
	"fmt"
	"path/filepath"

	"github.com/xxx/go-subburn/internal/task"
)

func BuildArgs(videoPath string, outPath string, cfg task.SubConfig) ([]string, error) {
	ffBin, err := GetFFmpegBinPath()
	if err != nil {
		return nil, err
	}
	args := []string{ffBin, "-y"}

	if cfg.Mode == "soft" {
		// 多路软字幕 mov_text
		args = append(args, "-i", videoPath)
		for _, tr := range cfg.Tracks {
			args = append(args, "-i", tr.File)
		}
		// 视频音频直接复制
		args = append(args, "-c:v", "copy", "-c:a", "copy")
		// 逐个字幕流设置编码 + metadata
		for idx, tr := range cfg.Tracks {
			streamTag := fmt.Sprintf("s:s:%d", idx)
			args = append(args,
				fmt.Sprintf("-c:%s", streamTag), "mov_text",
				fmt.Sprintf("-metadata:%s", streamTag), fmt.Sprintf("language=%s", tr.Lang),
				fmt.Sprintf("-metadata:%s", streamTag+":title"), tr.Title,
			)
		}
		args = append(args, outPath)
		return args, nil
	}

	if cfg.Mode == "burn" {
		// 硬字幕：只取第一条轨道烧录，支持ass/srt
		if len(cfg.Tracks) == 0 {
			return nil, fmt.Errorf("burn mode require at least one subtitle")
		}
		subFile := cfg.Tracks[0].File
		style := fmt.Sprintf("FontName=%s,FontSize=%d", cfg.FontName, cfg.FontSize)
		if cfg.FontFile != "" {
			style += fmt.Sprintf(",FontFile=%s", cfg.FontFile)
		}
		vf := fmt.Sprintf("subtitles=%s:force_style='%s'", filepath.ToSlash(subFile), style)
		args = append(args, "-i", videoPath)
		args = append(args, "-vf", vf, "-c:a", "copy")
		args = append(args, outPath)
		return args, nil
	}
	return nil, fmt.Errorf("unknown sub mode: %s", cfg.Mode)
}
```

## FFmpeg Runner `internal/ffmpeg/runner.go`
> 负责启动进程、读取 stderr、解析 `time=00:01:22.33` 进度、context 取消
```go
package ffmpeg

import (
	"bufio"
	"context"
	"fmt"
	"os/exec"
	"regexp"
	"strconv"
	"strings"

	"github.com/xxx/go-subburn/internal/task"
)

var timeRegex = regexp.MustCompile(`time=(\d+):(\d+):(\d+)\.(\d+)`)

type ProgressCallback func(currentSec float64, logLine string)

func RunJob(ctx context.Context, args []string, totalSec float64, cb ProgressCallback) error {
	cmd := exec.CommandContext(ctx, args[0], args[1:]...)
	stderr, err := cmd.StderrPipe()
	if err != nil {
		return err
	}
	if err := cmd.Start(); err != nil {
		return err
	}

	scanner := bufio.NewScanner(stderr)
	for scanner.Scan() {
		line := scanner.Text()
		cb(0, line)
		m := timeRegex.FindStringSubmatch(line)
		if m != nil {
			h, _ := strconv.Atoi(m[1])
			mi, _ := strconv.Atoi(m[2])
			s, _ := strconv.Atoi(m[3])
			ms, _ := strconv.Atoi(m[4])
			current := float64(h*3600+mi*60+s) + float64(ms)/100
			cb(current, "")
		}
	}
	err = cmd.Wait()
	if ctx.Err() == context.Canceled {
		return fmt.Errorf("cancelled")
	}
	return err
}
```

## `internal/ffmpeg/util.go` （获取ffmpeg路径 + 探测视频总时长）
```go
package ffmpeg

import (
	"embed"
	"fmt"
	"os"
	"runtime"
)

// 你可以embed或者直接读取build目录，这里简化
func GetFFmpegBinPath() (string, error) {
	root, _ := os.Getwd()
	var binName string
	switch runtime.GOOS {
	case "windows":
		binName = "ffmpeg.exe"
	case "darwin":
		binName = "ffmpeg"
	default:
		binName = "ffmpeg"
	}
	p := fmt.Sprintf("%s/build/ffmpeg/%s/%s", root, runtime.GOOS, binName)
	if _, err := os.Stat(p); err != nil {
		// fallback: 从PATH找
		return binName, nil
	}
	return p, nil
}
```

# 五、Wails 绑定入口 `app.go`
> Wails 把 Go 方法暴露给前端 Vue TS 调用：SubmitJob、CancelJob、GetJobs
```go
package main

import (
	"context"
	"fmt"
	"github.com/google/uuid"
	"github.com/wailsapp/wails/v2/pkg/runtime"
	"time"

	"github.com/xxx/go-subburn/internal/ffmpeg"
	"github.com/xxx/go-subburn/internal/task"
)

type App struct {
	ctx     context.Context
	jobMgr  *task.Manager
}

func NewApp() *App {
	return &App{
		jobMgr: task.NewManager(),
	}
}

func (a *App) Startup(ctx context.Context) {
	a.ctx = ctx
}

type SubmitJobReq struct {
	InputVideo  string        `json:"inputVideo"`
	OutputVideo string        `json:"outputVideo"`
	SubConfig   task.SubConfig `json:"subConfig"`
}

func (a *App) SubmitJob(req SubmitJobReq) (*task.Job, error) {
	jobId := uuid.NewString()
	ctxJob, cancel := context.WithCancel(context.Background())
	job := &task.Job{
		JobId:       jobId,
		InputVideo:  req.InputVideo,
		OutputVideo: req.OutputVideo,
		SubConfig:   req.SubConfig,
		Status:      "pending",
		ctx:         ctxJob,
		cancel:      cancel,
	}
	a.jobMgr.AddJob(job)

	// 异步跑任务
	go func() {
		job.Status = "running"
		args, err := ffmpeg.BuildArgs(job.InputVideo, job.OutputVideo, job.SubConfig)
		if err != nil {
			job.Status = "error"
			job.Log = append(job.Log, err.Error())
			a.jobMgr.UpdateJob(job)
			runtime.EventsEmit(a.ctx, "jobUpdate", job)
			return
		}
		runtime.EventsEmit(a.ctx, "jobUpdate", job)

		err = ffmpeg.RunJob(ctxJob, args, job.TotalSeconds, func(currentSec float64, logLine string) {
			if logLine != "" {
				job.Log = append(job.Log, logLine)
			}
			if currentSec > 0 {
				job.CurrentSeconds = currentSec
				if job.TotalSeconds > 0 {
					job.Progress = currentSec / job.TotalSeconds * 100
				}
			}
			runtime.EventsEmit(a.ctx, "jobUpdate", job)
		})

		if err != nil {
			if err.Error() == "cancelled" {
				job.Status = "cancelled"
			} else {
				job.Status = "error"
				job.Log = append(job.Log, err.Error())
			}
		} else {
			job.Status = "done"
			job.Progress = 100
		}
		a.jobMgr.UpdateJob(job)
		runtime.EventsEmit(a.ctx, "jobUpdate", job)
	}()
	return job, nil
}

func (a *App) CancelJob(jobId string) bool {
	return a.jobMgr.CancelJob(jobId)
}
```

> 前端订阅事件：`runtime.EventsOn("jobUpdate", (job)=>{...})` 刷新界面状态

# 七、关键注意事项（多路软字幕坑）
1. **`mov_text` 不支持 ASS**，多路软字幕只能 SRT；想要ASS特效字幕只能选硬烧录模式
2. 播放器兼容性：VLC、MPV、QuickTime 支持切换 mov_text 轨道；很多智能电视、网页播放器不识别多路 mov_text
3. 语言标签遵循 ISO 639-2：`zh` / `eng` / `jpn`
4. 流复制 `-c copy` 非常快，不重新编码视频，这是多路软字幕最大优势
5. 分发方案：把 ffmpeg 三个平台二进制放进 build，打包时通过 `wails.json` `Resources` 嵌入，运行时释放或者直接读取路径
